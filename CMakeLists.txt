cmake_minimum_required(VERSION 3.22)
project(SwellVocalLift VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(LIFT_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND LIFT_FORMATS AU)
endif()
juce_add_plugin(SwellVocalLift COMPANY_NAME "Swell-audio" BUNDLE_ID "fr.swell-audio.vocal-lift" PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE Svlf FORMATS ${LIFT_FORMATS} PRODUCT_NAME "Swell Vocal - Lift" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(VocalLiftAssets SOURCES swell-mark.png)
target_sources(SwellVocalLift PRIVATE Source/Plugin.cpp)
target_compile_definitions(SwellVocalLift PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SwellVocalLift PRIVATE VocalLiftAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
add_executable(VocalLiftDSPTest Tests/DSPTest.cpp)
add_test(NAME VocalLiftDSP COMMAND VocalLiftDSPTest)
